```python
import json

from read import packetize, verify_packets

tail = "Paragraphe de suite."
cases = {
    "double_backticks": ("Le code ``a`$$b`` est littéral.", False),
    "triple_backticks": ("Autre code ```a``$$b``` littéral.", False),
    "brackets_in_code": (r"Les signes `\[ et \]` sont cités.", False),
    "unpaired_backtick": ("Un backtick ` libre puis $$x=y$$.", True),
    "code_and_formula": ("Le signe `$` précède $$x=y$$.", True),
    "quoted_bracket_then_block": (r"On cite `\[` ici." + "\n\n" + "\\[\nx=y\n\\]", True),
}
report = {}
for name, (text, math) in cases.items():
    raw = (text + "\n## Suite\n" + tail + "\n").encode()
    doc = packetize(raw, name + ".md")
    expected = [(text, [], math), (tail, ["Suite"], False)]
    if name == "quoted_bracket_then_block":
        prose, block = text.split("\n\n")
        expected = [(prose, [], False), (block, [], True), (tail, ["Suite"], False)]
    actual = [(p["source_quote"], p["section_path"], p["contains_display_math"])
              for p in doc["packets"]]
    report[name] = {"packets": actual == expected,
                    "citations": not verify_packets(raw, doc)}
print(json.dumps(report, indent=2, sort_keys=True))
```
