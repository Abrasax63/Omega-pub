```python
from pathlib import Path
from tempfile import TemporaryDirectory
from unittest.mock import patch
import pipeline as bridge
from read import blob

a, b = b"Ancienne source publique.\n", b"Source publique remplacee.\n"
original = bridge.read_document
with TemporaryDirectory() as folder:
    source = Path(folder) / "public.md"
    source.write_bytes(a)
    def replace_after_inventory(path, sha):
        inventory = original(path, sha)
        path.write_bytes(b)
        return inventory
    with patch.object(bridge, "read_document", side_effect=replace_after_inventory):
        try:
            result = bridge.pipeline(source, blob(a), model_json={"source_blob": blob(b), "claims": []})
        except ValueError:
            print("REJECTED_SOURCE_CHANGED")
        else:
            print(result["source_blob"] == result["quarantine"]["source_blob"])
```
