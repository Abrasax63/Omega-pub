```python
import json

from read import packetize, verify_packets

cases = {
    "dollar_multiline": "$$\nx+y=z\n$$\n## Suite\nParagraphe de suite.\n",
    "bracket_multiline": "\\[\nx+y=z\n\\]\n## Suite\nParagraphe de suite.\n",
    "dollar_single_line": "$$x+y=z$$\n## Suite\nParagraphe de suite.\n",
    "plain": "Intro ordinaire.\n## Suite\nParagraphe de suite.\n",
}
report = {}
for name, source in cases.items():
    raw = source.encode("utf-8")
    doc = packetize(raw, f"{name}.md")
    packets = doc["packets"]
    tail = next(p for p in packets if "Paragraphe de suite." in p["source_quote"])
    checks = {
        "display_count": doc["stats"]["packets_with_display_math"] == (name != "plain"),
        "heading_not_in_packets": all("## Suite" not in p["source_quote"] for p in packets),
        "tail_section": tail["section_path"] == ["Suite"],
        "tail_not_math": not tail["contains_display_math"],
        "exact_source_spans": verify_packets(raw, doc) == [],
    }
    report[name] = {"checks": checks, "passed": all(checks.values())}
    if not report[name]["passed"]:
        report[name]["tail"] = {k: tail[k] for k in ("section_path", "source_quote", "contains_display_math")}
print(json.dumps(report, ensure_ascii=False, indent=2))
```
