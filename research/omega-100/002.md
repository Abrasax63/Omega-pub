```python
import json

from read import packetize, verify_packets

cases = {
    "blank_in_display": (
        "# Fiche\n$$\nx=y\n\nz=t\n$$\n## Retour\nPremier paragraphe.\n",
        1, {"Premier paragraphe.": ["Fiche", "Retour"]},
    ),
    "two_dollar_blocks": (
        "$$\na=b\n$$\n## Entre\nTexte intermédiaire.\n\n$$\nb=c\n$$\n## Après\nTexte final.\n",
        2, {"Texte intermédiaire.": ["Entre"], "Texte final.": ["Après"]},
    ),
    "two_delimiter_styles": (
        "\\[\na=b\n\\]\n## Seconde partie\nTexte distinct.\n\n$$\nb=c\n$$\n",
        2, {"Texte distinct.": ["Seconde partie"]},
    ),
    "ordinary_sections": (
        "# Début\n## Fin\nTexte sans formule.\n",
        0, {"Texte sans formule.": ["Début", "Fin"]},
    ),
}
report = {}
for name, (text, display_count, expected_sections) in cases.items():
    raw = text.encode("utf-8")
    doc = packetize(raw, f"{name}.md")
    prose = {p["source_quote"]: p["section_path"] for p in doc["packets"]
             if not p["contains_display_math"]}
    checks = {
        "display_count": doc["stats"]["packets_with_display_math"] == display_count,
        "prose_sections": prose == expected_sections,
        "exact_source_spans": verify_packets(raw, doc) == [],
    }
    report[name] = {"checks": checks, "passed": all(checks.values())}
print(json.dumps(report, ensure_ascii=False, indent=2))
```
